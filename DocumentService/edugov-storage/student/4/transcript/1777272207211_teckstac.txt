import React, { Component } from 'react';
import './styles.css';
import courseList from './courseList.json';

class Search extends Component {
    constructor(props) {
        super(props);
        this.state = {
            course: '',
            name: '',
            submitStatus: false
        };
    }

    displayName = (event) => {
        // Requirement: set the state for name by fetching text from the name text box
        this.setState({ name: event.target.value });
    }

    searchCourse = (event) => {
        const userInput = event.target.value;
        
        // Requirement: Compare irrespective of case
        // We use .some() to check if the exact string exists in the array
        const found = courseList.some(item => item.toLowerCase() === userInput.toLowerCase());

        let statusMessage = "";
        if (found) {
            // Exact Match for Screenshot 2
            statusMessage = `course '${userInput}' is currently available`;
        } else {
            // Exact Match for Screenshot 1
            statusMessage = `course '${userInput}' is currently not available`;
        }

        // Requirement: On entering the course name, 'on change' should take place
        this.setState({ course: statusMessage });
    }

    handleSubmit = (event) => {
        // Use preventDefault to avoid reset
        event.preventDefault();
        this.setState({ submitStatus: true });
    }

    render() {
        return (
            <div>
                <h2>EXCEL COACHING CENTRE</h2>
                <form onSubmit={this.handleSubmit}>
                    <label>Name</label><br />
                    <input type="text" id="name" onChange={this.displayName} /><br />

                    <label>Qualification</label><br />
                    <select>
                        <option value="BE/BTech">BE/BTech</option>
                        <option value="ME/MTech">ME/MTech</option>
                        <option value="BCA/MCA">BCA/MCA</option>
                    </select><br />

                    <label>Search by Course</label><br />
                    <input type="text" id="search" onChange={this.searchCourse} /><br />

                    <button type="submit" id="submit" onClick={this.handleSubmit}>Submit</button>
                </form>

                {/* Requirement: Display only when submitStatus is true */}
                {this.state.submitStatus && (
                    <Display 
                        name={this.state.name} 
                        course={this.state.course} 
                    />
                )}
            </div>
        );
    }
}

class Display extends Component {
    render() {
        // Requirement: "Welcome to Excel coaching centre!!! Hi <name>, <courseName>"
        // Note: <courseName> here refers to the statusMessage passed via props
        return (
            <div>
                <p>Welcome to Excel coaching centre!!!</p>
                <p>Hi {this.props.name}, {this.props.course}</p>
            </div>
        );
    }
}

export { Display };
export default Search;
